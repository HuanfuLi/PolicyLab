**Initialization:**
```typescript
useEffect(() => {
  const scenarioIds = searchParams.get('scenarios')?.split(',') ?? [id!];
  // Fetch session metadata for each to get scenarioLabel
  Promise.all(scenarioIds.map(sid => fetch(`/api/sessions/${sid}`).then(r => r.json())))
    .then(sessions => {
      const sessionInfos = sessions.map((s, i) => ({
        id: s.id,
        label: s.scenarioLabel ?? (i === 0 ? 'Baseline' : `Scenario ${String.fromCharCode(65 + i - 1)}`),
      }));
      multiScenarioStore.initScenarios(sessionInfos);
      multiScenarioStore.loadAllAgents();
      multiScenarioStore.loadAllHistory();
      const cleanup = multiScenarioStore.connectAll();
      return cleanup;
    });
}, [id, searchParams]);
```
