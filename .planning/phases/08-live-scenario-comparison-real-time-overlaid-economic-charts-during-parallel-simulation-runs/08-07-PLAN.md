1. In `server/src/llm/gateway.ts`, add a factory function for the load balancer:
```typescript
import { LoadBalancer } from './loadBalancer.js';
import type { ProviderConfig } from '@policylab/shared';

let loadBalancerInstance: LoadBalancer | null = null;

/** Get or create a load balancer that distributes across all configured providers.
 *  Falls back to a single-provider balancer using getProvider() if no providers array is configured. */
export function getLoadBalancer(): LLMProvider {
  if (loadBalancerInstance) return loadBalancerInstance;
  
  const settings = readSettings();
  if (settings.providers && settings.providers.length > 0) {
    const slots = settings.providers.map(pc => ({
      provider: createProviderFromSettings({ ...settings, provider: pc.provider, apiKey: pc.apiKey, baseUrl: pc.baseUrl ?? settings.baseUrl, centralAgentModel: pc.model ?? settings.centralAgentModel, vertexProjectId: pc.vertexProjectId ?? settings.vertexProjectId, vertexLocation: pc.vertexLocation ?? settings.vertexLocation }),
      label: `${pc.provider}/${pc.model ?? settings.centralAgentModel}`,
      rateLimit: pc.rateLimit,
    }));
    loadBalancerInstance = new LoadBalancer(slots);
  } else {
    // Single provider fallback
    loadBalancerInstance = new LoadBalancer([{
      provider: getProvider(),
      label: `${settings.provider}/${settings.centralAgentModel}`,
      rateLimit: null,
    }]);
  }
  return loadBalancerInstance;
}

// Add to invalidateProvider():
export function invalidateProvider(): void {
  provider = null;
  citizenProviderCache = null;
  loadBalancerInstance = null;
}
```
