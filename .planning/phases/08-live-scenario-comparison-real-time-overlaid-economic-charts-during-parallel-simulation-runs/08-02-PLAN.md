   ```typescript
   import type { LLMProvider, LLMMessage, LLMOptions, TestConnectionResult } from './types.js';

   class TokenBucket {
     private tokens: number;
     private lastRefill: number;
     constructor(private readonly rate: number, private readonly capacity: number) {
       this.tokens = capacity;
       this.lastRefill = Date.now();
     }
     tryConsume(): boolean {
       this.refill();
       if (this.tokens >= 1) { this.tokens -= 1; return true; }
       return false;
     }
     async waitForToken(): Promise<void> {
       while (!this.tryConsume()) {
         await new Promise(r => setTimeout(r, 50));
       }
     }
     private refill(): void {
       const now = Date.now();
       const elapsed = (now - this.lastRefill) / 1000;
       this.tokens = Math.min(this.capacity, this.tokens + elapsed * this.rate);
       this.lastRefill = now;
     }
   }

   interface ProviderSlot {
     provider: LLMProvider;
     label: string;
     bucket: TokenBucket | null;  // null = unlimited
     inflight: number;
   }

   export class LoadBalancer implements LLMProvider {
     private slots: ProviderSlot[];
     private index: number = 0;

     constructor(slots: Array<{ provider: LLMProvider; label: string; rateLimit?: number | null }>) {
       this.slots = slots.map(s => ({
         provider: s.provider,
         label: s.label,
         bucket: s.rateLimit != null ? new TokenBucket(s.rateLimit / 60, Math.ceil(s.rateLimit / 60)) : null,
         inflight: 0,
       }));
     }

     async chat(messages: LLMMessage[], options?: LLMOptions): Promise<string> {
       const slot = await this.acquireSlot();
       slot.inflight++;
       try { return await slot.provider.chat(messages, options); }
       finally { slot.inflight--; }
     }

     async *chatStream(messages: LLMMessage[], options?: LLMOptions): AsyncIterable<string> {
       const slot = await this.acquireSlot();
       slot.inflight++;
       try { yield* slot.provider.chatStream(messages, options); }
       finally { slot.inflight--; }
     }

     async testConnection(): Promise<TestConnectionResult> {
       return this.slots[0].provider.testConnection();
     }

     private async acquireSlot(): Promise<ProviderSlot> {
       // Try round-robin, skip rate-limited slots
       for (let attempt = 0; attempt < this.slots.length; attempt++) {
         const slot = this.slots[this.index % this.slots.length];
         this.index++;
         if (!slot.bucket || slot.bucket.tryConsume()) return slot;
       }
       // All rate-limited: wait on next slot in round-robin order
       const slot = this.slots[this.index % this.slots.length];
       this.index++;
       if (slot.bucket) await slot.bucket.waitForToken();
       return slot;
     }
   }
   ```
