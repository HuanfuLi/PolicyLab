1. **Create `shared/src/scenarioDataMerge.ts` (placed in shared so it can be unit-tested via server's vitest config)** — the data merging utility:
```typescript
import type { TelemetryLog, IterationStats } from '@policylab/shared';

export interface ScenarioMeta {
  label: string;
  index: number;  // 0-3, determines color/dash
  sessionId: string;
}

export interface MergedDataPoint {
  iterationNumber: number;
  [key: string]: number | null | undefined;
}

/** Merge N scenario TelemetryLog arrays into flat rows for recharts.
 *  Each row: { iterationNumber, "field_Baseline": 102.3, "field_Policy A": 108.7, ... } */
export function mergeScenarioData(
  scenarios: Array<{ label: string; data: TelemetryLog[] }>,
  field: keyof TelemetryLog
): MergedDataPoint[] {
  const byIteration = new Map<number, MergedDataPoint>();
  for (const { label, data } of scenarios) {
    for (const row of data) {
      const key = `${String(field)}_${label}`;
      const existing = byIteration.get(row.iterationNumber) ?? { iterationNumber: row.iterationNumber };
      existing[key] = (row[field] as number) ?? null;
      byIteration.set(row.iterationNumber, existing);
    }
  }
  return Array.from(byIteration.values()).sort((a, b) => a.iterationNumber - b.iterationNumber);
}

/** Merge N scenario IterationStats arrays for agent stat charts.
 *  Uses the same pattern as mergeScenarioData but for IterationStats fields. */
export function mergeStatsData(
  scenarios: Array<{ label: string; data: IterationStats[] }>,
  field: keyof IterationStats
): MergedDataPoint[] {
  const byIteration = new Map<number, MergedDataPoint>();
  for (const { label, data } of scenarios) {
    for (const row of data) {
      const key = `${String(field)}_${label}`;
      const existing = byIteration.get(row.iterationNumber) ?? { iterationNumber: row.iterationNumber };
      existing[key] = (row[field] as number) ?? null;
      byIteration.set(row.iterationNumber, existing);
    }
  }
  return Array.from(byIteration.values()).sort((a, b) => a.iterationNumber - b.iterationNumber);
}
```

```typescript
import { describe, it, expect } from "vitest";
import { mergeScenarioData, mergeStatsData } from "@policylab/shared/scenarioDataMerge";

describe("mergeScenarioData", () => {
  it("merges 2 scenarios into flat rows keyed by field_label", () => {
    const scenarios = [
      { label: "Baseline", data: [{ iterationNumber: 1, cpiIndex: 100 }, { iterationNumber: 2, cpiIndex: 102 }] as any[] },
      { label: "Policy A", data: [{ iterationNumber: 1, cpiIndex: 100 }, { iterationNumber: 2, cpiIndex: 108 }] as any[] },
    ];
    const result = mergeScenarioData(scenarios, "cpiIndex" as any);
    expect(result).toHaveLength(2);
    expect(result[0]["cpiIndex_Baseline"]).toBe(100);
    expect(result[1]["cpiIndex_Policy A"]).toBe(108);
  });

  it("handles missing iterations (sparse data) with null fill", () => {
    const scenarios = [
      { label: "A", data: [{ iterationNumber: 1, cpiIndex: 100 }] as any[] },
      { label: "B", data: [{ iterationNumber: 2, cpiIndex: 110 }] as any[] },
    ];
    const result = mergeScenarioData(scenarios, "cpiIndex" as any);
    expect(result).toHaveLength(2);
    expect(result[0]["cpiIndex_B"]).toBeUndefined();
    expect(result[1]["cpiIndex_A"]).toBeUndefined();
  });

  it("returns empty array for empty input", () => {
    expect(mergeScenarioData([], "cpiIndex" as any)).toEqual([]);
  });
});
```

```tsx
import { ChevronLeft, ChevronRight } from 'lucide-react';

interface CollapsiblePanelProps {
  side: 'left' | 'right';
  collapsed: boolean;
  onToggle: () => void;
  label: string;  // "Live Feed" or "Agent Status"
  children: React.ReactNode;
}

export function CollapsiblePanel({ side, collapsed, onToggle, label, children }: CollapsiblePanelProps) {
  const Icon = side === 'left' ? ChevronRight : ChevronLeft;  // expand direction
  const CollapseIcon = side === 'left' ? ChevronLeft : ChevronRight;
  
  return (
    <div style={{
      width: collapsed ? 40 : undefined,
      minWidth: collapsed ? 40 : 300,
      transition: 'width 0.3s ease, min-width 0.3s ease',
      overflow: 'hidden',
      position: 'relative',
      background: 'var(--glass-bg)',
      border: '1px solid var(--glass-border)',
      borderRadius: 10,
    }}>
      {collapsed ? (
        <button onClick={onToggle} title={`Expand ${label}`} style={{
          position: 'absolute', top: '50%', left: '50%',
          transform: 'translate(-50%, -50%)',
          background: 'none', border: 'none', cursor: 'pointer',
          color: 'var(--text-dim)',
        }}>
          <Icon size={20} />
        </button>
      ) : (
        <>
          <button onClick={onToggle} style={{
            position: 'absolute', top: 8,
            [side === 'left' ? 'right' : 'left']: 8,
            background: 'none', border: 'none', cursor: 'pointer',
            color: 'var(--text-dim)', zIndex: 1,
          }}>
            <CollapseIcon size={16} />
          </button>
          {children}
        </>
      )}
    </div>
  );
}
```

```tsx
interface ScenarioProgressBarProps {
  label: string;
  current: number;
  total: number;
  color: string;
  isComplete: boolean;
  isPaused: boolean;
  error: string | null;
}
export function ScenarioProgressBar({ label, current, total, color, isComplete, isPaused, error }: ScenarioProgressBarProps) {
  const pct = total > 0 ? Math.round((current / total) * 100) : 0;
  return (
    <div style={{ flex: '1 1 200px', minWidth: 150 }}>
      <div style={{ fontSize: 12, fontWeight: 700, color, marginBottom: 2 }}>
        {label}: {error ? 'Error' : isComplete ? 'Complete' : `${pct}%`}
      </div>
      <div style={{ height: 8, borderRadius: 4, background: 'var(--panel-alpha-10, rgba(255,255,255,0.1))' }}>
        <div style={{
          width: `${pct}%`, height: '100%', borderRadius: 4,
          background: error ? 'var(--danger)' : color,
          opacity: isPaused ? undefined : 0.8,
          animation: isPaused ? 'pulse 1.5s ease-in-out infinite' : undefined,
        }} />
      </div>
    </div>
  );
}
```
