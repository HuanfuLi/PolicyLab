    ```typescript
    import type { TaxPolicy } from '@policylab/shared';

    export type TaxKind = 'wage' | 'amm_sell' | 'vat' | 'capital_gains';

    /**
     * Compute tax to withhold from a fiat-bearing event.
     * Phase 11 D-12, D-14.
     *
     * @param income Gross amount subject to tax (positive fiat). Caller has already determined the taxable base.
     * @param kind   Which tax rate applies. 'wage' and 'amm_sell' use rates.income; 'vat' uses rates.vat; 'capital_gains' uses rates.capitalGains.
     * @param policy Session's TaxPolicy. If undefined (legacy session), returns 0.
     */
    export function computeWithholding(
      income: number,
      kind: TaxKind,
      policy: TaxPolicy | undefined,
    ): number {
      if (!policy || income &lt;= 0) return 0;

      // Dispatch to rate by kind
      const pickFlatRate = (): number =&gt; {
        switch (kind) {
          case 'wage':
          case 'amm_sell':
            return policy.rates.income;
          case 'vat':
            return policy.rates.vat;
          case 'capital_gains':
            return policy.rates.capitalGains;
        }
      };

      if (policy.kind === 'flat') {
        return income * clampRate(pickFlatRate());
      }

      // progressive
      if (kind === 'vat' || kind === 'capital_gains') {
        // Progressive brackets apply only to income-style events; VAT and capital gains stay at flat rate
        return income * clampRate(pickFlatRate());
      }

      // kind === 'wage' | 'amm_sell': piecewise-progressive
      const brackets = policy.brackets ?? [];
      if (brackets.length === 0) {
        return income * clampRate(policy.rates.income);   // fallback to top rate
      }
      let tax = 0;
      let remaining = income;
      let prevUpto = 0;
      for (const b of brackets) {
        if (remaining &lt;= 0) break;
        const bracketWidth = Math.max(0, b.upto - prevUpto);
        const portion = Math.min(remaining, bracketWidth);
        tax += portion * clampRate(b.rate);
        remaining -= portion;
        prevUpto = b.upto;
      }
      if (remaining &gt; 0) {
        // Over the top bracket: use policy.rates.income as the top marginal rate
        tax += remaining * clampRate(policy.rates.income);
      }
      return tax;
    }

    function clampRate(r: number): number {
      return Math.max(0, Math.min(0.5, r));    // CONTEXT.md per-rate clamp [0, 0.5]
    }
    ```
