    ```typescript
    import type { TaxPolicy } from '@policylab/shared';

    const DEFAULT_TAX_POLICY: TaxPolicy = {
      kind: 'flat',
      rates: { income: 0.15, vat: 0.10, capitalGains: 0.15 },
    };

    function clampRate(r: unknown): number {
      const n = typeof r === 'number' ? r : 0;
      return Math.max(0, Math.min(0.5, n));
    }

    export function validateTaxPolicy(input: unknown): TaxPolicy {
      if (!input || typeof input !== 'object') return DEFAULT_TAX_POLICY;
      const raw = input as Partial&lt;TaxPolicy&gt;;
      if (raw.kind !== 'flat' &amp;&amp; raw.kind !== 'progressive') return DEFAULT_TAX_POLICY;
      const rates = {
        income: clampRate((raw.rates as any)?.income),
        vat: clampRate((raw.rates as any)?.vat),
        capitalGains: clampRate((raw.rates as any)?.capitalGains),
      };
      if (raw.kind === 'flat') {
        return { kind: 'flat', rates };
      }
      // progressive
      const brackets = Array.isArray(raw.brackets) ? raw.brackets : [];
      if (brackets.length === 0) return { ...DEFAULT_TAX_POLICY, rates };
      // Validate strictly increasing upto
      let prev = -Infinity;
      for (const b of brackets) {
        if (typeof b?.upto !== 'number' || b.upto &lt;= prev) {
          return { kind: 'flat', rates };   // malformed → coerce to flat with same rates
        }
        prev = b.upto;
      }
      const cleanBrackets = brackets.map(b =&gt; ({ upto: b.upto, rate: clampRate(b.rate) }));
      return { kind: 'progressive', rates, brackets: cleanBrackets };
    }
    ```
