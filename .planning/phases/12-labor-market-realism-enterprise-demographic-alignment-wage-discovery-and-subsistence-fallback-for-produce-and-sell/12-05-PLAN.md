       ```typescript
       // ── Phase 12 D-13: APPLY_FOR_JOB automated matching pass ─────────────────
       {
         const reservationWages = sessionReservationWages.get(sessionId) ?? new Map<string, number>();
         const minWage = iterEconomyConfig.minimumWage ?? 5;
         const reservationFloor = Math.max(minWage * 0.5, 1);  // D-12 fallback; small_constant = 1

         // Build applicant pool: APPLY_FOR_JOB this iteration + QUIT last iteration (auto-reapply)
         const applicantIds = new Set<string>();
         for (const intent of intents) {   // intents is the per-iteration structured intent list
           if (intent.actions?.some(a => a.actionCode === 'APPLY_FOR_JOB')) {
             applicantIds.add(intent.agentId);
           }
         }
         const quitSet = sessionQuitLastIteration.get(sessionId) ?? new Set<string>();
         for (const agentId of quitSet) applicantIds.add(agentId);
         sessionQuitLastIteration.set(sessionId, new Set());  // consume-and-clear

         // Snapshot applicant counts per enterprise BEFORE clearing
         const applicantCountsByEnt = new Map<string, number>();
         for (const ent of enterpriseRegistry.values()) {
           applicantCountsByEnt.set(ent.id, ent.applicants.size);
         }

         // Greedy best-offer matching
         let placementsThisIter = 0;
         for (const agentId of applicantIds) {
           if (employmentRegistry.has(agentId)) continue;
           const reservationWage = reservationWages.get(agentId) ?? reservationFloor;
           const qualifying = [...enterpriseRegistry.values()]
             .filter(ent => ((ent.capacity ?? 0) - ent.employees.size) > 0 && ent.wage >= reservationWage)
             .sort((a, b) => b.wage !== a.wage ? b.wage - a.wage : a.employees.size - b.employees.size);
           if (qualifying.length === 0) continue;
           const best = qualifying[0]!;
           employmentRegistry.set(agentId, {
             enterpriseId: best.id,
             employerId: best.ownerId,
             employeeId: agentId,
             wage: best.wage,
             minSkill: best.minSkill,
             startedAt: iterationNumber,   // adapt to local variable name
           });
           best.employees.add(agentId);
           const agentState = weekStateMap.get(agentId);
           if (agentState) agentState.employer_id = best.id;
           placementsThisIter++;
         }

         // Persist lastApplicants / lastVacancies for next-iter wage nudge (D-03)
         for (const ent of enterpriseRegistry.values()) {
           ent.lastApplicants = applicantCountsByEnt.get(ent.id) ?? 0;
           ent.lastVacancies = Math.max(0, (ent.capacity ?? 0) - ent.employees.size);
           ent.applicants.clear();
         }

         appendTrace(
           sessionId,
           `[MATCH] Matching pass: ${applicantIds.size} applicants, ${placementsThisIter} placed, ` +
           `${[...enterpriseRegistry.values()].reduce((s,e) => s + e.lastVacancies, 0)} vacancies remaining`,
         );
       }
       ```
