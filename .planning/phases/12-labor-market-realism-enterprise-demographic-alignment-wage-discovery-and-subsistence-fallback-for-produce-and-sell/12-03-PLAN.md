      overview: string;
      agentRoster: readonly { role: string; sector?: EnterpriseSector; type?: string }[];
      baseFiat: number;
      minimumWage: number;
      /** Optional healing feedback appended to the user message for retry attempts. */
      healingFeedback?: string;
    }

    export function buildEnterpriseBlueprintsPrompt(
      params: EnterpriseBlueprintPromptParams,
    ): { system: string; user: string } {
      const { overview, agentRoster, baseFiat, minimumWage, healingFeedback } = params;

      // Compute employable bucket summary
      const employable = agentRoster.filter(a =>
        a.type?.toLowerCase() !== 'bank' &&
        !['central_bank','central_agent','official'].includes((a.role ?? '').toLowerCase())
      );
      const sectorCounts: Record<string, number> = {};
      for (const a of employable) {
        const s = a.sector ?? 'unspecified';
        sectorCounts[s] = (sectorCounts[s] ?? 0) + 1;
      }
      const employableCount = employable.length;
      const targetVacancies = Math.ceil(employableCount * 1.10);
      const wageAnchor = Math.max(minimumWage, baseFiat * 0.10);

      const system = 'You are generating the enterprise roster for a PolicyLab simulation society. ' +
        'Output must be valid JSON matching the schema exactly; nothing else.';

      const sectorBreakdown = Object.entries(sectorCounts)
        .map(([s, n]) => `  - ${s}: ${n} employable citizens (${Math.round(100 * n / employableCount)}%)`)
        .join('\n');

      const schemaBlock = `JSON schema:
[
  {
    "sector": "agriculture" | "industry" | "services" | "government",
    "ownerRole": string,        // e.g. "farm_owner", "factory_owner", "shop_owner", "administrator"
    "initialWorkforceSize": number,  // starting employee count at this enterprise
    "capacity": number,         // max workforce (initialWorkforceSize + open vacancies)
    "wageAnchor": number        // initial posted wage in fiat; suggested default ${wageAnchor.toFixed(2)}
  }
]`;

      const invariants = `INVARIANTS (must hold over the full enterprise list):
1. Every sector with ≥5% of the ${employableCount} employable citizens MUST have ≥1 enterprise.
2. SUM of capacity across all enterprises MUST be ≥ ${targetVacancies} (=110% of employable count).
3. Each enterprise capacity ≥ 2 and ≥ initialWorkforceSize.`;

      const healing = healingFeedback
        ? `\n\nPREVIOUS ATTEMPT FAILED VALIDATION:\n${healingFeedback}\n\nRegenerate the list so all invariants hold.`
        : '';

      const user = `Society overview:\n${overview}\n\nEmployable citizens by sector:\n${sectorBreakdown}\n` +
        `\nTotal employable: ${employableCount}\n\n${schemaBlock}\n\n${invariants}${healing}`;

      return { system, user };
    }
    ```

    import type { AgentBlueprint, EnterpriseBlueprint, EnterpriseSector } from '@policylab/shared';
    import { retryWithHealing } from '../llm/retryWithHealing.js';
    import { buildEnterpriseBlueprintsPrompt } from '../llm/prompts/central-agent.js';
    import { isEmployableAgent } from '../orchestration/helpers/isEmployableAgent.js';

    // Chat client type — accept whatever existing shape the central-agent calls use.
    // If the project exposes an LLMClient interface, import and use that directly.
    export interface EnterpriseLLMClient {
      complete: (system: string, user: string) => Promise<string>;
    }

    interface ValidatorInput {
      employableCount: number;
      sectorCounts: Map<EnterpriseSector | string, number>;
    }

    function validateEnterpriseRoster(
      raw: string,
      { employableCount, sectorCounts }: ValidatorInput,
    ): { ok: true; value: EnterpriseBlueprint[] } | { ok: false; healing: string } {
      let parsed: unknown;
      try { parsed = JSON.parse(raw); } catch (e) {
        return { ok: false, healing: `Output was not valid JSON: ${(e as Error).message}` };
      }
      if (!Array.isArray(parsed)) {
        return { ok: false, healing: 'Top-level JSON must be an array of enterprise blueprints.' };
      }

      // Shape validation: each entry has required fields
      const blueprints: EnterpriseBlueprint[] = [];
      for (const [i, item] of (parsed as unknown[]).entries()) {
        if (typeof item !== 'object' || item === null) {
          return { ok: false, healing: `Entry ${i}: must be an object.` };
        }
        const it = item as Record<string, unknown>;
        const sector = it.sector as EnterpriseSector;
        if (!['agriculture','industry','services','government'].includes(sector)) {
          return { ok: false, healing: `Entry ${i}: sector must be agriculture|industry|services|government.` };
        }
        const capacity = typeof it.capacity === 'number' ? it.capacity : Number(it.capacity);
        if (!Number.isFinite(capacity) || capacity < 2) {
          return { ok: false, healing: `Entry ${i}: capacity must be a number ≥ 2.` };
        }
        blueprints.push({
          id: `ent_${sector.slice(0,4)}_${i+1}`,
          name: `${sector} enterprise ${i+1}`,
          ownerId: String(it.ownerRole ?? 'owner'),
          sector,
          industry: String(it.sector),
          commodityOutput: sector === 'agriculture' ? 'food' : sector === 'industry' ? 'tools' : sector === 'services' ? 'luxury_goods' : 'none',
          initialCapital: 0,
          initialInventory: {},
          employees: [],
          wage: typeof it.wageAnchor === 'number' ? it.wageAnchor : 5,
          isServiceEnterprise: sector === 'government',
          capacity,
        });
      }

      // Invariant 1: each ≥5% sector has ≥1 enterprise
      const coveredSectors = new Set(blueprints.map(b => b.sector));
      const missing: string[] = [];
      for (const [sector, count] of sectorCounts) {
        if (count / employableCount >= 0.05 && !coveredSectors.has(sector as EnterpriseSector)) {
          missing.push(String(sector));
        }
      }

      // Invariant 2: sum(capacity) ≥ 1.10 × employable
      const sumCapacity = blueprints.reduce((s, b) => s + (b.capacity ?? 0), 0);
      const target = Math.ceil(employableCount * 1.10);

      if (missing.length > 0 || sumCapacity < target) {
        const covered = [...coveredSectors].join(', ') || '(none)';
        const healing = `The roster you generated covers sectors [${covered}] but sectors [${missing.join(', ') || '(none)'}] have ≥5% workforce with zero employers. Total capacity: ${sumCapacity}. Need ≥ ${target}. Add enterprises for the missing sectors and/or raise capacity so every sector with ≥5% workforce has ≥1 enterprise and sum(capacity) ≥ 1.10 × ${employableCount}.`;
        return { ok: false, healing };
      }

      return { ok: true, value: blueprints };
    }

    export async function generateEnterprisesFromCentralAgent(params: {
      overview: string;
      agentRoster: AgentBlueprint[];
      baseFiat: number;
      minimumWage: number;
      llm: EnterpriseLLMClient;
    }): Promise<EnterpriseBlueprint[]> {
      const { overview, agentRoster, baseFiat, minimumWage, llm } = params;

      const employable = agentRoster.filter(isEmployableAgent);
      const sectorCounts = new Map<EnterpriseSector | string, number>();
      for (const a of employable) {
        const s = (a.sector ?? 'unspecified') as EnterpriseSector | string;
        sectorCounts.set(s, (sectorCounts.get(s) ?? 0) + 1);
      }
      const validatorInput: ValidatorInput = { employableCount: employable.length, sectorCounts };

      let lastFeedback: string | undefined = undefined;

      return retryWithHealing(
        async () => {
          const { system, user } = buildEnterpriseBlueprintsPrompt({
            overview, agentRoster, baseFiat, minimumWage,
            healingFeedback: lastFeedback,
          });
          return llm.complete(system, user);
        },
        (raw: string) => {
          const result = validateEnterpriseRoster(raw, validatorInput);
          if (!result.ok) lastFeedback = result.healing;
          return result;
        },
        { maxAttempts: 3 },
      );
    }
    ```
