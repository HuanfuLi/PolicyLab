         const wealth = Array(nAgents).fill(50);
         const tools  = Array(nAgents).fill(5);
         const netByIter: number[][] = [];
         for (let it = 0; it < iters; it++) {
           const iterNet: number[] = [];
           for (let a = 0; a < nAgents; a++) {
             const produced = tools[a] > 0 ? 4 : Math.floor(4 * 0.3);  // raw-material shortage fallback
             const consumed = 2;  // MET baseline
             const sellable = Math.max(0, produced - consumed);
             // AMM sell using x·y=k

             const revenue = amm.fiatReserve - newFiat;
